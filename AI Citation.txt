import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
import folium
from folium.plugins import MarkerCluster, HeatMap
from pygbif import occurrences as occ

# Set visual style
sns.set_theme(style="whitegrid")
plt.rcParams['figure.figsize'] = (10, 5)

# ================= ==============================================
# 1. FETCH DATA FROM GBIF API (Mid-Atlantic Bounding Box)
# ================================================================
# Bounding box coordinates covering NC/VA border up through NJ/PA
LAT_RANGE = "36.5,41.5"
LON_RANGE = "-80.5,-74.0"

print("Fetching species occurrences in the Mid-Atlantic region...")

# Request up to 300 records with valid coordinates
data = occ.search(
    decimalLatitude=LAT_RANGE,
    decimalLongitude=LON_RANGE,
    hasCoordinate=True,
    limit=300
)

records = data['results']
df = pd.DataFrame(records)

# Clean and select essential columns
columns_needed = ['scientificName', 'species', 'family', 'decimalLatitude', 'decimalLongitude', 'year', 'basisOfRecord']
existing_cols = [col for col in columns_needed if col in df.columns]
df_clean = df[existing_cols].dropna(subset=['decimalLatitude', 'decimalLongitude']).copy()

print(f"Successfully loaded {len(df_clean)} records.\n")
print(df_clean.head())

# ================= ==============================================
# 2. EXPLORATORY DATA CHARTS
# ================================================================

# Chart 1: Top 10 Most Frequently Recorded Species
fig, ax = plt.subplots(figsize=(10, 5))
top_species = df_clean['species'].value_counts().head(10)
sns.barplot(x=top_species.values, y=top_species.index, palette="viridis", ax=ax)
ax.set_title("Top 10 Most Recorded Species in the Mid-Atlantic Search", fontsize=14, fontweight='bold')
ax.set_xlabel("Observation Count")
ax.set_ylabel("Species")
plt.tight_layout()
plt.show()

# Chart 2: Records by Data Type (Basis of Record)
if 'basisOfRecord' in df_clean.columns:
    fig, ax = plt.subplots(figsize=(8, 4))
    record_types = df_clean['basisOfRecord'].value_counts()
    sns.barplot(x=record_types.index, y=record_types.values, palette="mako", ax=ax)
    ax.set_title("Distribution of Records by Basis of Record", fontsize=14, fontweight='bold')
    ax.set_xlabel("Record Type")
    ax.set_ylabel("Count")
    plt.xticks(rotation=15)
    plt.tight_layout()
    plt.show()

# Chart 3: Temporal Distribution (Observations per Year)
if 'year' in df_clean.columns:
    df_years = df_clean.dropna(subset=['year']).copy()
    df_years['year'] = df_years['year'].astype(int)
    
    fig, ax = plt.subplots(figsize=(10, 4))
    sns.histplot(df_years['year'], bins=15, kde=True, color="#2b5c8f", ax=ax)
    ax.set_title("Distribution of Occurrence Records Over Time", fontsize=14, fontweight='bold')
    ax.set_xlabel("Year")
    ax.set_ylabel("Count")
    plt.tight_layout()
    plt.show()

# ================= ==============================================
# 3. INTERACTIVE GEOGRAPHIC MAP (Folium)
# ================================================================

# Center map on the Mid-Atlantic region
map_center = [39.0, -77.2]
m = folium.Map(location=map_center, zoom_start=7, tiles="Cartodb Positron")

# Layer 1: Clustered Point Markers
marker_cluster = MarkerCluster(name="Clustered Records").add_to(m)

for idx, row in df_clean.iterrows():
    species_name = row.get('species', row.get('scientificName', 'Unknown Species'))
    year_val = row.get('year', 'N/A')
    
    popup_text = f"<b>Species:</b> {species_name}<br><b>Year:</b> {year_val}"
    
    folium.Marker(
        location=[row['decimalLatitude'], row['decimalLongitude']],
        popup=folium.Popup(popup_text, max_width=250),
        tooltip=species_name,
        icon=folium.Icon(color="teal", icon="leaf", prefix="fa")
    ).add_to(marker_cluster)

# Layer 2: Density Heatmap
heat_data = [[row['decimalLatitude'], row['decimalLongitude']] for idx, row in df_clean.iterrows()]
HeatMap(heat_data, name="Observation Density (Heatmap)", show=False).add_to(m)

# Layer Controls to toggle layers on/off
folium.LayerControl().add_to(m)

# Display map directly in Jupyter Notebook

Solution: Resample Rasters to Match NLCD Grid
You need to reproject and resample your climate rasters (bio1, bio6, bio12, bio15) so their resolution, bounding box, and dimensions match nlcd_landcover.tif exactly.

Replace your dataset loading/extraction code before Cell 10 with this rasterio.reproject pipeline:

Python
import numpy as np
import rasterio
from rasterio.enums import Resampling
from rasterio.warp import reproject, calculate_default_transform

# 1. Read NLCD reference raster (Target Grid)
with rasterio.open(landcover_path) as nlcd_src:
    nlcd_data = nlcd_src.read(1)
    target_crs = nlcd_src.crs
    target_transform = nlcd_src.transform
    target_shape = nlcd_src.shape  # (height, width)

def read_and_resample(raster_path, target_shape, target_crs, target_transform):
    """Resamples any raster to match NLCD's shape, CRS, and extent."""
    with rasterio.open(raster_path) as src:
        resampled_array = np.empty(target_shape, dtype=np.float32)
        reproject(
            source=rasterio.band(src, 1),
            destination=resampled_array,
            src_transform=src.transform,
            src_crs=src.crs,
            dst_transform=target_transform,
            dst_crs=target_crs,
            resampling=Resampling.bilinear
        )
        return resampled_array

# 2. Load and resample all climate variables to match NLCD dimensions
bio1_data = read_and_resample(clim_files['Bio1'], target_shape, target_crs, target_transform)
bio6_data = read_and_resample(clim_files['Bio6'], target_shape, target_crs, target_transform)
bio12_data = read_and_resample(clim_files['Bio12'], target_shape, target_crs, target_transform)
bio15_data = read_and_resample(clim_files['Bio15'], target_shape, target_crs, target_transform)

# 3. Stack features (all arrays now have size 25,000,000)
X_present = np.column_stack([
    bio1_data.ravel(),
    bio6_data.ravel(),

Why this happens in GBIF data:
County Centroid Default: Observers sometimes enter only the County name (e.g., Accomack County, VA), and automated geocoders place the lat/lon directly in the geographical center of the county polygon—which often falls into tidal marshes, bays, or open ocean.

Precision Loss: Geocoordinates rounded to 1 decimal place can shift terrestrial observations miles out into coastal waters.
